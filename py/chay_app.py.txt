# Cầu nối giữa trang web và bộ quy tắc: nhận 1 đơn (JSON), trả danh sách vấn đề (JSON)
import json, sys
sys.path.insert(0, "/rs/engine")
import ra_soat as R, thoi_diem as TD, ghi_chu_ds as G, dinh_chi as DC
DL = "/rs/du_lieu/"
_C = {}
def _nap():
    if not _C:
        _C["dm"] = R.doc_danh_muc(DL + "Danh_muc_thong_tin_thuoc.xlsx")
        _C["Q"] = R.doc_quy_tac(DL + "Bang_tra_cuu_ra_soat.xlsx")
        _C["td"] = TD.doc_bang(DL + "Thoi_diem_dung_thuoc.xlsx")
    return _C
def ten_danh_muc():
    return sorted({str(v.get("ten")) for v in _nap()["dm"].values() if v.get("ten")})
def ra_soat_don(don_json):
    c = _nap(); d = json.loads(don_json)
    don = dict(lk=d.get("lk") or "DON-ANH", tuoi=d.get("tuoi"), bac_si=d.get("bac_si") or "", ngay=d.get("ngay") or "",
               icd_chinh=[x.strip().upper() for x in d.get("icd_chinh") or [] if x.strip()],
               icd_kem=[x.strip().upper() for x in d.get("icd_kem") or [] if x.strip()],
               thuoc=[dict(ma=str(i), ten=t["ten"].strip(), dv=t.get("dv") or "", sl=t.get("sl"), lieu=t.get("lieu") or "")
                      for i, t in enumerate(d.get("thuoc") or []) if (t.get("ten") or "").strip()])
    issues, ct = R.ra_soat([don], c["dm"], c["Q"], c["td"])
    G.ap_dung(issues, ct, DL + "Ghi_chu_DS.xlsx")
    keep_i = ("muc", "tieu_chi", "thuoc", "hoat_chat", "van_de", "de_xuat", "can_cu", "nhan_xet", "xuat_toan", "muc_tt", "qd5948")
    keep_c = ("thuoc", "hoat_chat", "ham_luong", "so_luong", "don_vi", "cach_dung", "dv_ngay", "so_ngay", "phu_hop_icd", "can_cu_icd", "danh_muc", "thoi_diem_hdsd", "nguon_hdsd")
    return json.dumps(dict(issues=[{k: i.get(k) for k in keep_i} for i in issues], chi_tiet=[{k: x.get(k) for k in keep_c} for x in ct]),
                      ensure_ascii=False, default=str)
