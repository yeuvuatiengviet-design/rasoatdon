def ud(s):
    s = unicodedata.normalize("NFD", str(s or "")).replace("đ", "d").replace("Đ", "D")
    return "".join(c for c in s if unicodedata.category(c) != "Mn").lower()

def norm(s): return re.sub(r"\s+", " ", ud(s).strip())

def hc_gon(s):
    """'Perindopril tert-butylamin 4 mg (tương đương ...) + Amlodipin (dưới dạng ...) 5 mg' -> 'Perindopril tert-butylamin + Amlodipin'"""
    s = str(s or "")
    for _ in range(3): s = re.sub(r"\([^()]*\)", "", s)
    s = re.sub(r"\d+(?:[.,]\d+)?\s*(?:mg|mcg|µg|g|iu|ui|ml|%)\b", "", s, flags=re.I)
    s = re.sub(r"\s*[;+]\s*", " + ", s); s = re.sub(r"\s{2,}", " ", s).strip(" +;,")
    return s[:120]

def split_list(s): return [x.strip() for x in str(s or "").replace(";", ",").split(",") if x.strip()]

def icd_match(code, spec):
    code = code.upper()
    for t in split_list(spec):
        t = t.upper()
        if t == "*": return t
        if "-" in t:
            a, b = t.split("-", 1)
            c = code[:len(a)]
            if a <= c <= b[:len(a)] or (c[:len(b)] <= b and c >= a): return t
        elif code.startswith(t): return t
    return None

def any_icd(codes, spec):
    out = []
    for c in codes:
        m = icd_match(c, spec)
        if m: out.append(c)
    return out

ICD_RE = re.compile(r"\b([A-Z]\d{2}(?:\.\d{1,2})?)")

def doc_bao_cao(path):
    ws = openpyxl.load_workbook(path, read_only=True, data_only=True).worksheets[0]
    rows = list(ws.iter_rows(values_only=True))
    hi = next(i for i, r in enumerate(rows) if any(ud(c).strip() == "ma duoc pham" for c in r if c))
    H = {ud(c).strip(): j for j, c in enumerate(rows[hi]) if c}
    col = lambda *names: next(H[n] for n in names if n in H)
    cMa, cTen, cDv, cSl, cLieu = col("ma duoc pham"), col("ten duoc pham"), col("don vi"), col("so luong"), col("lieu dung")
    cCd, cIcd, cBenh, cKem = col("chan doan"), col("ma icd"), col("ten benh chinh"), col("benh kem theo")
    dons, cur = [], None
    for r in rows[hi + 1:]:
        g = lambda j: r[j] if j < len(r) else None
        dv = str(g(cDv) or "")
        if "LK:" in dv or "Kê ngày" in dv:
            m = re.search(r"^(.*?)\s*Kê ngày\s*(\d{2}/\d{2}/\d{4}).*?LK:\s*(\S+)", dv)
            bs, ngay, lk = (m.group(1).strip(), m.group(2), m.group(3)) if m else ("", "", dv)
            dob = re.search(r"(\d{2}/\d{2}/\d{4})", str(g(cTen) or ""))
            tuoi = None
            try:
                d0 = datetime.datetime.strptime(dob.group(1), "%d/%m/%Y"); d1 = datetime.datetime.strptime(ngay, "%d/%m/%Y")
                tuoi = d1.year - d0.year - ((d1.month, d1.day) < (d0.month, d0.day))
            except Exception: pass
            ho_ten = re.split(r"\s*/\s*\d{1,2}/\d{1,2}/\d{4}", str(g(cTen) or ""))[0].strip(" /")
            icd_chinh = ICD_RE.findall(str(g(cIcd) or ""))
            icd_kem = [x for x in ICD_RE.findall(str(g(cKem) or "")) if x not in icd_chinh]
            cur = dict(lk=lk, ho_ten=ho_ten, ma_bn=str(g(cMa) or ""), bac_si=bs, ngay=ngay, tuoi=tuoi, chan_doan=str(g(cCd) or ""), icd_chinh=icd_chinh,
                       icd_kem=icd_kem, ten_benh_kem=str(g(cKem) or ""), thuoc=[])
            dons.append(cur)
        elif cur is not None and g(cTen) and g(cMa):
            cur["thuoc"].append(dict(ma=str(g(cMa)), ten=str(g(cTen)).strip(), dv=dv, sl=g(cSl), lieu=str(g(cLieu) or "")))
    return dons

def doc_danh_muc(path):
    wb = openpyxl.load_workbook(path, read_only=True, data_only=True)
    ws = next((w for w in wb.worksheets if "danh muc" in ud(w.title)), wb.worksheets[0])
    rows = [r for r in ws.iter_rows(values_only=True)]
    hdr = [ud(c).strip() if c else "" for c in rows[0]]
    def idx(key): return next((i for i, h in enumerate(hdr) if key in h), None)
    I = dict(ten=idx("ten thuoc"), hc=idx("hoat chat"), hl=idx("ham luong"), dang=idx("dang bao che"), duong=idx("duong dung"),
             cd=idx("chi dinh"), lieu=idx("lieu dung"), ccd=idx("chong chi dinh"), tt=idx("than trong"), tuongtac=idx("tuong tac"),
             link=idx("to huong dan"), gc=idx("ghi chu"))
    dm = {}
    for r in rows[1:]:
        if len(r) <= I["ten"] or not r[I["ten"]]: continue
        dm[norm(r[I["ten"]])] = {k: (r[v] if v is not None and v < len(r) else None) for k, v in I.items()}
    return dm

def nguon_hdsd(info):
    """Đánh giá độ tin cậy cột Chỉ định của danh mục dựa trên link/ghi chú đối chiếu."""
    if not info: return ""
    link = str(info.get("link") or ""); gc = ud(info.get("gc") or "")
    if not link: return "Danh mục chưa có tờ HDSD gốc"
    if "drive.google" not in link: return "Nguồn web, không phải tờ HDSD gốc"
    # Link Drive cũng có thể chứa PDF ghép từ ảnh web hoặc tài liệu thông tin thuốc (không phải tờ HDSD) → dựa vào ghi chú
    if any(k in gc for k in ("tai lieu thong tin", "khong phai to hdsd")):
        return "Tài liệu thông tin thuốc, không phải tờ HDSD – xem cột Ghi chú DM"
    if any(k in gc for k in ("nguon web", "danh may lai", "khong phai ban scan")):
        return "Nguồn web (PDF trên Drive), không phải tờ HDSD gốc"
    if any(k in gc for k in ("chu y", "luu y", "khong phai", "khac ham luong", "thieu", "khong doc duoc", "can kiem tra")):
        return "Ghi chú đối chiếu có lưu ý – xem cột Ghi chú DM"
    return "Tờ HDSD gốc"

def tim_danh_muc(dm, ten):
    k = norm(ten)
    if k in dm: return dm[k], "khớp tên"
    k2 = k.replace(" ", "")
    for kk, v in dm.items():
        if kk.replace(" ", "") == k2: return v, "khớp tên"
    m = difflib.get_close_matches(k, dm.keys(), 1, 0.85)
    return (dm[m[0]], "gần đúng: " + m[0]) if m else (None, "không có trong danh mục")

def tra_bang(bang, *ten):
    """Tra theo tên đơn thuốc, tên trong danh mục; khớp bỏ dấu cách; gần đúng ≥0.92."""
    for t in ten:
        if not t: continue
        k = norm(t)
        if k in bang: return bang[k]
        k2 = k.replace(" ", "")
        for kk, v in bang.items():
            if kk.replace(" ", "") == k2: return v
    for t in ten:
        if not t: continue
        m = difflib.get_close_matches(norm(t), bang.keys(), 1, 0.92)
        if m: return bang[m[0]]
    return None

def gioi_han_lieu(sp, tuoi, icds):
    """Trả (max đơn vị/ngày, mô tả) theo tuổi và ICD; None nếu không áp dụng."""
    if not sp: return None, ""
    for spec, v in sp.get("theo_icd") or []:
        if any_icd(icds, spec): return v, f"theo ICD {spec}"
    tnl = sp.get("tuoi_nl") or 12
    if tuoi is not None and tuoi < tnl:
        return (sp.get("max_tre"), f"trẻ <{tnl:g} tuổi") if sp.get("max_tre") is not None else (None, "")
    return sp.get("max_nl"), "người lớn"

    """Trả (min đơn vị/ngày, mô tả) cho người lớn ≥18 tuổi; None nếu không áp dụng."""

_DEM = r"vien|virn|goi|ong|v\b|g\b"
DV_MAP = {"vien": _DEM, "goi": _DEM, "ong": _DEM}
def don_vi_ngay(lieu, dv):
    t = ud(lieu).replace(",", ".")
    t = re.sub(r"(\d)\s*/\s*(\d)", r"\1/\2", t)          # "1/ 2 viên" -> "1/2"
    if re.search(r"khi (sot|dau|can|ho|kho tho)|neu (sot|dau)", t): return None, "dùng khi cần"
    key = ud(dv).strip().split()[0] if dv else ""
    pat = DV_MAP.get(key)
    if not pat: return None, "đơn vị không tính được"
    num = r"(?<!\d)(?<!\d\.)(\d+(?:\.\d+)?(?:/\d+)?)(?!\d)(?!\.\d)(?!/\d)"
    def f(x):
        if "/" in x: a, b = x.split("/"); return float(a) / float(b)
        return float(x)
    m = re.search(num + r"\s*(?:" + pat + r")\s*[x×*]\s*(\d+)\s*lan", t)
    if m: return f(m.group(1)) * int(m.group(2)), ""
    m = re.search(r"(\d+)\s*lan.{0,20}?moi lan\s*" + num + r"\s*(?:" + pat + ")", t)
    if m: return int(m.group(1)) * f(m.group(2)), ""
    m = re.search(r"ngay\s*(?:uong|dung|ngam)?\s*" + num + r"\s*(?:" + pat + ")", t)
    if m: return f(m.group(1)), ""
    vals = [f(x) for x in re.findall(num + r"\s*(?:" + pat + ")", t)]
    if len(vals) >= 3 and abs(vals[0] - sum(vals[1:])) < 1e-9: return vals[0], ""   # "uống 2 viên: sáng 1, tối 1"
    slots = [f(x) for x in re.findall(r"(?:sang|trua|chieu|toi|\d{1,2}\s*h)\s*(?:uong|ngam)?\s*:?\s*" + num + r"(?!\s*(?:h\b|ml|mg|gio|phut|lan|p\b|ui|iu|dv|don vi|giot))", t)]
    if re.search(r"\d\s*(?:ui|iu|dv|don vi)\b", t): slots = []
    if len(slots) >= 2 and len(slots) > len(vals): return sum(slots), ""        # "sáng 2 tối 2 viên", "sáng 1 tói 1 viên"
    if vals: return sum(vals), ""
    # dạng viết tắt: "Sáng 1", "sáng 1 tối 1", "7h 1"
    vals = [f(x) for x in re.findall(r"(?:sang|trua|chieu|toi|\d{1,2}\s*h)\s*(?:uong|ngam)?\s*" + num + r"(?!\s*(?:h\b|ml|mg|gio|phut|lan|p\b))", t)]
    if vals: return sum(vals), ""
    m = re.search(r"ngay\s*(?:uong|dung)?\s*" + num + r"(?!\s*(?:lan|l\b|ngay|tuan|h\b|gio|phut|ml|mg|p\b|giot))", t)   # "uống ngày 01 trước ăn"
    if m: return f(m.group(1)), ""
    return None, "không đọc được số lượng/lần"

def mg_moi_don_vi(hl):
    s = ud(hl)
    if not s or "+" in s or ";" in s or "/ml" in s or "/g" in s: return None
    m = re.match(r"\s*(\d+(?:[.,]\d+)?)\s*(mg|g|mcg|µg|ug)\b", s.replace("µ", "u"))
    if not m: return None
    v = float(m.group(1).replace(".", "").replace(",", ".")) if re.match(r"^\d{1,3}\.\d{3}$", m.group(1)) else float(m.group(1).replace(",", "."))
    return v * {"mg": 1, "g": 1000, "mcg": 0.001, "ug": 0.001, "µg": 0.001}[m.group(2)]

FILL = {"Nghiêm trọng": "F8CBAD", "Cần lưu ý": "FFE699", "Tham khảo": "DDEBF7",
        "Không phù hợp": "F8CBAD", "Không rõ": "FFE699", "Phù hợp có điều kiện": "FFF2CC", "Phù hợp": "E2EFDA", "Chưa có quy tắc": "EDEDED", "Ngoài HDSD": "F4B183", "Chưa đối chiếu HDSD": "EDEDED"}

def ghi_sheet(wb, title, cols, data, color_key=None):
    ws = wb.create_sheet(title)
    ws.append([c[1] for c in cols])
    for d in data:
        ws.append([d.get(c[0]) for c in cols])
    for i, c in enumerate(cols, 1): ws.column_dimensions[get_column_letter(i)].width = c[2]
    for c in ws[1]:
        c.font = Font(bold=True, color="FFFFFF"); c.fill = PatternFill("solid", fgColor="1F4E78")
        c.alignment = Alignment(wrap_text=True, vertical="center")
    ki = [c[0] for c in cols].index(color_key) + 1 if color_key else None
    for row in ws.iter_rows(min_row=2):
        for c in row: c.alignment = Alignment(wrap_text=True, vertical="top")
        if ki and row[ki - 1].value in FILL: row[ki - 1].fill = PatternFill("solid", fgColor=FILL[row[ki - 1].value])
    ws.freeze_panes = "A2"; ws.auto_filter.ref = ws.dimensions
    return ws
